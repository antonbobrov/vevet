## CursorHoverElement

Runtime representation of a hoverable element attached to the cursor.
Created by **[attachHover()](methods#attachhover)** and exposed via  
**[hoveredElement](accessors#hoveredelement)** while that target is active.

```ts
class CursorHoverElement {
  /** Hoverable DOM element */
  get element(): Element;

  /** Events emitter (defaults to `element`) */
  get emitter(): Element;

  /** Cursor type applied for this element */
  get type(): string | undefined;

  /** Whether the cursor snaps to the element center */
  get snap(): boolean;

  /** Target cursor width while hovering */
  get width(): null | number | 'auto';

  /** Target cursor height while hovering */
  get height(): null | number | 'auto';

  /** Padding applied around the cursor */
  get padding(): number;

  /** Whether sticky parallax is enabled */
  get sticky(): boolean;

  /** Sticky pull-back friction (`0`–`1`) */
  get friction(): number;

  /** Whether sticky friction is active */
  get hasFriction(): boolean;

  /**
   * @deprecated Use attachHover `ease` / cursor ease instead.
   */
  get stickyLerp(): number | undefined;
}
```

## ICursorHoverElementProps

Configuration object passed to **[attachHover()](methods#attachhover)**  
to define cursor behavior for a specific hoverable element.

```ts
interface ICursorHoverElementProps {
  /** Hoverable DOM element */
  element: Element;

  /**
   * Hover events emitter.
   * If not provided, the element itself will be used.
   * Useful when the visual element and the hover area are different nodes.
   * @default the element itself
   */
  emitter?: Element | null;

  /** Cursor type to activate on hover */
  type?: string;

  /**
   * Debounce time for hover events, in milliseconds
   * @default 16
   */
  hoverDebounce?: number;

  /**
   * Snap cursor to the element center.
   * Does not work with `behavior: "path"`.
   * @default false
   */
  snap?: boolean;

  /**
   * Target cursor width on hover.
   * Supports css units like `px`, `rem`, `vw`, `vh`, `svh`.
   * @default null
   */
  width?: null | number | 'auto' | (string & {});

  /**
   * Target cursor height on hover.
   * Supports css units like `px`, `rem`, `vw`, `vh`, `svh`.
   * @default null
   */
  height?: null | number | 'auto' | (string & {});

  /**
   * Padding applied around the cursor.
   * Supports css units like `px`, `rem`, `vw`, `vh`, `svh`.
   * @default 0
   */
  padding?: number | string;

  /**
   * Enable sticky behavior for the hovered element.
   * @default false
   */
  sticky?: boolean;

  /**
   * Ease for sticky offset interpolation (damp `number` or spring).
   * @default Cursor `ease` / `lerp`
   */
  ease?: number | { stiffness: number; damping: number };

  /**
   * @deprecated Use `ease` instead.
   */
  stickyLerp?: number;

  /**
   * Sticky pull-back toward the origin while friction is active (`0`–`1`).
   * Higher values snap back faster. `0` disables friction.
   * @default 0
   */
  friction?: number;

  /**
   * @deprecated Use `friction` instead.
   */
  stickyFriction?: number;

  /**
   * Sticky parallax amplitude (uniform or per axis).
   * `'auto'` uses the hovered element size on that axis.
   * Supports css units like `px`, `rem`, `vw`, `vh`, `svh`.
   * @default 'auto'
   */
  amp?:
    | TCursorHoverElementStickyAmplitude
    | TCursorHoverElementStickyAmplitudeObject;

  /**
   * @deprecated Use `amp` instead.
   */
  stickyAmplitude?:
    | TCursorHoverElementStickyAmplitude
    | TCursorHoverElementStickyAmplitudeObject;
}
```

## TCursorHoverElementStickyAmplitude

Defines the amplitude of the sticky animation.

```ts
type TCursorHoverElementStickyAmplitude = number | 'auto' | (string & {});
```

## TCursorHoverElementStickyAmplitudeObject

Axis-based sticky amplitude configuration.

```ts
type TCursorHoverElementStickyAmplitudeObject = {
  x: TCursorHoverElementStickyAmplitude;
  y: TCursorHoverElementStickyAmplitude;
};
```

## ICursorType

Custom cursor type configuration for **[attachCursor()](methods#attachcursor)**.

```ts
interface ICursorType {
  /** Custom cursor element */
  element: Element;

  /** Unique cursor type identifier */
  type: string;
}
```
